-- ============================================================================
-- ISS-S3-003 · Semana 3
-- Da al caso un lugar donde vivir su explicacion y los datos extraidos del
-- documento de identidad. Ambas cosas se producen DESPUES de abrir el caso, de
-- forma asincrona, asi que el esquema debe representar explicitamente el estado
-- "abierto pero todavia sin explicar" — no como ausencia de dato, sino como un
-- estado consultable.
-- ============================================================================

-- --------------------------------------------------------------------------
-- 1. Correlacion y explicacion en el caso
-- --------------------------------------------------------------------------

-- La cuenta viajaba en flagged-case-v1 pero se perdia al abrir el caso: quedaba
-- solo dentro del texto libre de auditoria. Sin ella no se puede contrastar el
-- documento de identidad con el titular.
ALTER TABLE fraud_case ADD COLUMN IF NOT EXISTS account_id VARCHAR(128);

-- Contexto de traza W3C heredado de la ingesta. Es lo que permite saltar del
-- caso a la traza completa de la transaccion que lo origino.
ALTER TABLE fraud_case ADD COLUMN IF NOT EXISTS traceparent VARCHAR(64);

ALTER TABLE fraud_case ADD COLUMN IF NOT EXISTS explanation TEXT;

-- PENDING  : caso abierto, explicacion aun no generada (estado normal al abrir)
-- GENERATED: explicacion disponible
-- FAILED   : el explicador agoto sus reintentos; el caso sigue siendo valido
ALTER TABLE fraud_case ADD COLUMN IF NOT EXISTS explanation_state VARCHAR(16) NOT NULL DEFAULT 'PENDING';

ALTER TABLE fraud_case ADD COLUMN IF NOT EXISTS explained_at TIMESTAMPTZ;
ALTER TABLE fraud_case ADD COLUMN IF NOT EXISTS explanation_attempts INTEGER NOT NULL DEFAULT 0;

ALTER TABLE fraud_case DROP CONSTRAINT IF EXISTS ck_fraud_case_explanation_state;
ALTER TABLE fraud_case ADD CONSTRAINT ck_fraud_case_explanation_state
    CHECK (explanation_state IN ('PENDING', 'GENERATED', 'FAILED'));

-- El explicador consulta exactamente "casos pendientes, mas antiguo primero".
-- El indice parcial solo cubre esas filas: las ya explicadas son la mayoria y no
-- aportan nada a esta consulta.
CREATE INDEX IF NOT EXISTS idx_fraud_case_pending_explanation
    ON fraud_case (opened_at)
    WHERE explanation_state = 'PENDING';

CREATE INDEX IF NOT EXISTS idx_fraud_case_account ON fraud_case (account_id);

-- --------------------------------------------------------------------------
-- 2. Documentos de verificacion adjuntos al caso
-- --------------------------------------------------------------------------

-- Un documento ilegible NO puede interrumpir el flujo ni dejar el caso en estado
-- indeterminado. Por eso el resultado del intento de extraccion es una fila con
-- estado propio: el fallo queda registrado y consultable, no perdido en un log.
CREATE TABLE IF NOT EXISTS case_verification_document (
    id                        BIGSERIAL PRIMARY KEY,
    case_id                   BIGINT NOT NULL,
    blob_path                 VARCHAR(512) NOT NULL,
    content_type              VARCHAR(128),
    document_state            VARCHAR(32) NOT NULL,
    extracted_full_name       VARCHAR(256),
    extracted_document_number VARCHAR(64),
    extracted_birth_date      DATE,
    extracted_expiry_date     DATE,
    extraction_confidence     NUMERIC(4,3),
    extraction_engine         VARCHAR(64),
    failure_reason            TEXT,
    analyst_notified_at       TIMESTAMPTZ,
    received_at               TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    processed_at              TIMESTAMPTZ,
    CONSTRAINT uq_case_verification_document_blob UNIQUE (blob_path),
    CONSTRAINT fk_case_verification_document_case
        FOREIGN KEY (case_id) REFERENCES fraud_case(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT ck_case_verification_document_state
        CHECK (document_state IN (
            'RECEIVED',
            'EXTRACTED',
            'PARTIALLY_EXTRACTED',
            'UNREADABLE',
            'UNSUPPORTED_FORMAT',
            'EXTRACTION_FAILED'))
);

CREATE INDEX IF NOT EXISTS idx_case_verification_document_case
    ON case_verification_document (case_id);

CREATE INDEX IF NOT EXISTS idx_case_verification_document_pending
    ON case_verification_document (received_at)
    WHERE document_state = 'RECEIVED';

COMMENT ON TABLE case_verification_document IS
    'Resultado del intento de extraccion documental. Un fallo de extraccion es una fila con estado, no una ausencia de fila: el analista debe poder consultar que paso con el documento que cargo.';

COMMENT ON COLUMN fraud_case.explanation_state IS
    'PENDING mientras el explicador no ha corrido. Con el explicador detenido los casos siguen abriendose en este estado y se explican al restablecerlo.';
